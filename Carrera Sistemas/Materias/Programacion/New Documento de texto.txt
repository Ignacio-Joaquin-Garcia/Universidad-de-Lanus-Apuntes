void ordenarMatriz(int f,int c,int matriz[f][c])
{
    float aux;
    for(int i=0; i<f; i++)
    {
        for(int j=0; j<c-1; j++)
        {
            for(int k=0; k<c-j-1; k++)
            {
                if(matriz[i][k]>matriz[i][k+1])
                {
                    aux=matriz[i][k];
                    matriz[i][k]=matriz[i][k+1];
                    matriz[i][k+1]=aux;
                }
            }
        }

    }
};

{
    int encontrado=0;
